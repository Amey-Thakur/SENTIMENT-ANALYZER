from textblob import TextBlob from pyodide.ffi import create_proxy import js import asyncio # Update Progress Bar for User Feedback async def update_loader(percent, status): js.document.getElementById("progress-bar").style.width = f"{percent}%" js.document.getElementById("loader-status").innerHTML = status await asyncio.sleep(0.1) async def finalize_loader(): await update_loader(100, "Inference Engine Standby") await asyncio.sleep(0.5) js.document.getElementById("loader-wrapper").classList.add("fade-out") async def handle_analysis(): user_text = js.document.getElementById("user-text").value.strip() result_div = js.document.getElementById("result-overlay") output_badge = js.document.getElementById("sentiment-output") meta_p = js.document.getElementById("analysis-meta") spinner = js.document.getElementById("loading-spinner") if not user_text: return result_div.classList.add("hidden") spinner.classList.remove("hidden") await asyncio.sleep(0.5) try: blob = TextBlob(user_text) polarity = blob.sentiment.polarity if polarity < 0: sentiment="Negative" color_class="negative-text" elif polarity==0: sentiment="Neutral" color_class="neutral-text" else: sentiment="Positive" color_class="positive-text" output_badge.innerHTML=sentiment output_badge.className=f"sentiment-badge {color_class}" meta_p.innerHTML=f"Semantic Polarity Detection: {round(polarity, 4)}" spinner.classList.add("hidden") result_div.classList.remove("hidden") result_div.classList.add("pop-in") except Exception as e: js.console.error(f"Analysis error: {str(e)}") spinner.classList.add("hidden") # Execute Loader Transition on Start asyncio.ensure_future(finalize_loader())
